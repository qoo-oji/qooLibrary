## 2.2 エラー型

```swift
public protocol UserPresentableError: Error, Sendable {
    /// 何が起きたか（1 文） [ER-03]
    var whatHappened: String { get }
    /// なぜ起きたか [ER-03]
    var whyItHappened: String { get }
    /// 次に何ができるか（0..n 個の提案） [ER-03]
    var recoverySuggestions: [RecoveryAction] { get }
    /// 折りたたんで表示する技術詳細 [ER-03]
    var technicalDetail: String? { get }
    /// 提示強度 [13.1 節]
    var severity: NotificationSeverity { get }
}

public enum NotificationSeverity: Int, Sendable, Comparable {
    case logOnly = 5        // 強度 5: ログのみ [ER-01]
    case transient = 4      // 強度 4: 一時通知（ステータスバー／トースト）
    case inline = 3         // 強度 3: インライン
    case sheet = 2          // 強度 2: シート（判断が必要）
    case appModal = 1       // 強度 1: アプリモーダル（続行不可のみ） [ER-02]
}

public struct RecoveryAction: Sendable, Identifiable {
    public let id: String
    public let title: String
    public let kind: Kind
    public enum Kind: Sendable {
        case openWindow(WindowRoute)     // 例: 孤立ファイル整理へ [NT-05]
        case runCommand(ManualCommandID) // 例: 再スキャン [MX-xx]
        case openSystemSettings(String)
        case retry
        case dismiss
    }
}
```

### 領域別エラー

| 型 | 領域 | 代表ケース |
|---|---|---|
| `FileOperationError` | ファイル操作 | 権限不足、I/O、衝突未解決、オフライン |
| `ArchiveError` | 圧縮・展開 | 未対応形式、パスワード付き [AR-08]、展開爆弾 [EX-21]、パス封じ込め違反 [EX-11] |
| `FormatCompileError` | フォーマット定義 | 括弧不一致、`@title` 重複、同一 `@labelgroup#` 重複 [FF-16]、自由文字列連続 [TY-05] |
| `ScanError` | スキャン | ブックマーク解決失敗、ボリューム未接続 |
| `PersistenceError` | 永続化 | 移行失敗 [MG-11]、ダウングレード [MG-12]、ストア破損 [RB-03] |
| `ImportExportError` | JSON | スキーマ版が新しすぎる [IE-14]、パース失敗 |
| `LockError` | 排他 | `.busy(target: String, processName: String)` [LK-14] |

```swift
// 続行不可のみが強度 1 [ER-02]
extension PersistenceError: UserPresentableError {
    public var severity: NotificationSeverity {
        switch self {
        case .storeCorrupted, .migrationFailed, .schemaTooNew: return .appModal
        default: return .transient
        }
    }
}
```

## 2.3 通知の提示ルーティング [ER-01]

エラー／通知の提示方法を機能ごとに決めさせない。すべて `NotificationRouter` を通す。

```swift
@MainActor
public final class NotificationRouter {
    public func present(_ item: NotificationItem, in scene: SceneContext?) async -> RecoveryAction?
    /// 一括処理向け。判断が必要なものだけを都度尋ね、残りは集約する [ER-10]
    public func beginBatch(title: String) -> BatchNotificationSession
}

public struct NotificationItem: Sendable, Identifiable {
    public let id: UUID
    public let date: Date
    public let category: Category          // .error / .warning / .info [NT-03]
    public let severity: NotificationSeverity
    public let target: NotificationTarget? // ライブラリ / ファイル / 処理名 [NT-04]
    public let title: String
    public let body: String
    public let technicalDetail: String?
    public let actions: [RecoveryAction]
    public let operationLogID: UUID?       // 操作履歴へのリンク [NT-04]
}

public final class BatchNotificationSession {
    /// 判断が必要なものだけシートで尋ねる。"以降すべてに適用" を必ず持つ [ER-11][FM-12]
    public func ask(_ decision: PendingDecision) async -> DecisionOutcome
    /// 判断不要な失敗は記録のみ。処理は中断しない [ER-12][ER-13]
    public func record(failure: UserPresentableError, item: String)
    public func record(skip reason: String, item: String)
    public func record(success item: String)
    /// 完了後に "成功 N / 失敗 N / スキップ N" と理由別内訳を提示 [ER-14]
    public func finish() async -> BatchSummary
}

public struct BatchSummary: Sendable {
    public let succeeded: Int
    public let failed: [(item: String, error: any UserPresentableError)]
    public let skipped: [(item: String, reason: String)]
    public var failureBreakdown: [String: Int]      // 理由別件数 [ER-14]
    public var retryableItems: [String]             // [ER-15]
}
```

| # | 要件 |
|---|---|
| CB-10 | `present` は `severity` に応じて提示手段を自動選択する。呼び出し側が提示手段を指定できない API 設計とする [ER-01] |
| CB-11 | `severity >= .transient`（強度 4 以上）のすべてを `NotificationHistoryStore` に記録する [ER-05][NT-01] |
| CB-12 | 同種のエラーが連続する場合、`(category, code, target.libraryID)` をキーに 5 秒窓で集約し、件数付き 1 件として扱う [ER-22] |
| CB-13 | 部分成功はロールバックしない。`BatchSummary` を返して終わる [ER-16][MV-12][CW-35] |

## 2.4 通知履歴ストア [13.2 節]

```swift
public protocol NotificationHistoryStore: Sendable {
    func append(_ item: NotificationItem) async
    func query(category: NotificationItem.Category?,
               libraryID: UUID?,
               period: DateInterval?,
               keyword: String?) async -> [NotificationItem]     // [NW-05]
    func unreadCount() async -> Int                              // [NT-02]
    func markRead(_ ids: [UUID]) async
    func markAllRead() async                                     // [NW-03]
    func delete(_ ids: [UUID]) async
    func deleteAll() async
    func purgeExpired(retentionDays: Int, maxCount: Int) async    // [NT-07]
    func exportCSV() async throws -> Data                         // [NW-07]
}
```

- 実体は SwiftData の `NotificationRecord`（07 章 §7.2）。
- ステータスバーのバッジは**この未読件数のみ**を表示する。孤立ファイル [OR-05]、ペンディング [PW-14]、再スキャン警告 [MX-10] の個別バッジは持たない [NT-06]。

## 2.7 操作履歴 [18.1 節]

```swift
public struct OperationLogEntry: Sendable, Identifiable {
    public let id: UUID
    public let date: Date
    public let commandName: String            // "12 件のファイルを保管庫に移す"
    public let kind: OperationKind            // .fileMove / .rename / .labelAssign / …
    public let targets: [OperationTarget]     // 変更前後のパス・ラベル
    public let libraryID: UUID?
    public let undone: Bool
    public let detail: [String: String]       // 変換リネームの前後名など [CR-53]
}

public protocol OperationHistoryStore: Sendable {
    func record(_ entry: OperationLogEntry) async               // [HS-01][HS-03]
    func query(keyword: String?, period: DateInterval?, kind: OperationKind?) async -> [OperationLogEntry]
    func exportCSV() async throws -> Data                       // [HS-02]
    func purgeExpired(retentionDays: Int) async                 // [HS-04]
}
```

`FileOperationService` と `CommandStack` が記録の起点となる [FO-03]。ユースケース側からの直接記録は行わない [設計判断: 記録漏れを構造的に防ぐ]。
