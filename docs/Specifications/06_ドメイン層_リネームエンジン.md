## 6.1 自動リネーム [12.2 節]

### 6.1.1 ルールのモデル

```swift
public struct AutoRenameRule: Sendable, Hashable, Codable, Identifiable {
    public let id: UUID
    public var name: String
    public var isEnabled: Bool                 // [AR2-05]
    public var priority: Int                   // 登録順＝適用順 [AR2-04]
    public var target: Target                  // 対象
    public var condition: Condition            // 条件
    public var action: Action                  // 動作
    public var trimAdjacentWhitespace: Bool    // 削除時に直前・直後の空白も削除
    public var exceptions: [RuleException]     // AND 評価 [AR2-23]

    public enum Target: Sendable, Hashable, Codable {
        case fileExtension
        case namePrefix
        case nameSuffix
        case nameAnywhere
    }
    public enum Condition: Sendable, Hashable, Codable {
        case exact(String)
        case prefix(String)
        case suffix(String)
        case contains(String)
        case pattern(String)     // `??` = 任意桁数の数字。それ以外はリテラル
    }
    public enum Action: Sendable, Hashable, Codable {
        case replace(with: String)
        case delete
        case surround(before: String, after: String)
    }
}

public enum RuleException: Sendable, Hashable, Codable {
    case nameContains(String)          // [AR2-20]
    case nameNotContains(String)       // [AR2-21]
    case inSubfolder(relativePath: String)  // [AR2-22]
}
```

### 6.1.2 評価

```swift
public enum AutoRenameEngine {
    /// マッチしたルールをすべて順次適用する [AR2-04]
    public static func apply(_ name: String,
                             extension ext: String,
                             relativeFolder: String,
                             rules: [AutoRenameRule],
                             globalExceptions: [RuleException]) -> RenameOutcome

    public struct RenameOutcome: Sendable {
        public let newName: String
        public let newExtension: String
        public let appliedRuleIDs: [UUID]
        public var changed: Bool { … }
    }
}
```

```pseudo
apply(name, ext, folder, rules, globalExceptions):
    if any(globalExceptions matches (name, folder)): return unchanged   # [AR2-24]
    cur = name; curExt = ext; applied = []
    for r in rules where isEnabled sortedBy priority:
        if any(r.exceptions matches (cur, folder)): continue            # [AR2-23]
        switch r.target:
          case .fileExtension:
              if matches(r.condition, curExt): curExt = transform(curExt, r.action)
          case .namePrefix:
              range = 先頭からの一致範囲（r.condition）
          case .nameSuffix:
              range = 末尾までの一致範囲
          case .nameAnywhere:
              range = 最初の一致範囲（全出現ではなく最初の 1 箇所）
        if range != nil:
            cur = applyAction(cur, range, r.action, trim: r.trimAdjacentWhitespace)
            applied.append(r.id)
    return RenameOutcome(cur, curExt, applied)
```

| # | 要件 |
|---|---|
| AR3-01 | `.nameAnywhere` は**最初の一致 1 箇所のみ**を置換する [設計判断: 全置換は非冪等になりやすく、FO-21 と衝突するため。全置換が必要なら同じルールを複数登録する] |
| AR3-02 | `pattern` の `??` は「1 桁以上の数字（半角・全角）」にマッチする |
| AR3-03 | `trimAdjacentWhitespace` が真の場合、削除範囲の直前・直後の空白（半角・全角）を 1 連続分まとめて削除する |
| AR3-04 | 拡張子の変更は `.fileExtension` ルールでのみ行う。変換リネームでは拡張子を変えない [CR-06] |

### 6.1.3 冪等性検証 [FO-21〜FO-24]

```swift
public enum IdempotencyChecker {
    public enum Verdict: Sendable {
        case idempotent                                   // f(f(x)) == f(x)
        case convergesAfter(Int)                          // n 回で不動点。警告 [FO-22]
        case diverges(samples: [String])                  // 保存拒否 [FO-22]
    }
    /// ルール保存時に呼ぶ。サンプルはユーザーが入力したプレビュー用文字列 +
    /// ルール自身の出力例 + 既定のサンプル集。
    public static func verify(_ rules: [AutoRenameRule], samples: [String]) -> Verdict
}
```

```pseudo
verify(rules, samples):
    worst = .idempotent
    for s in samples:
        x0 = apply(rules, s)
        x1 = apply(rules, x0)
        if x1 == x0: continue
        # 収束するか調べる
        seen = [s, x0, x1]; x = x1; n = 2
        while n < AppLimits.idempotencyProbeIterations:
            x = apply(rules, x); n += 1
            if x == seen.last: worst = max(worst, .convergesAfter(n)); break
            if seen.contains(x): return .diverges([… 循環の証跡 …])   # 循環 → 保存拒否
            seen.append(x)
        else:
            return .diverges(seen)
    return worst
```

| # | 要件 |
|---|---|
| ID2-01 | `.diverges` は**保存を拒否**する [FO-22] |
| ID2-02 | `.convergesAfter(n)` は警告を表示し、保存は許可する [FO-22] |
| ID2-03 | 実行前に不動点チェックを行う。現在名 == 適用後の名前ならスキップする [FO-23] |
| ID2-04 | ID2-01〜03 により、FSEvents の帰属判定が失敗しても無限リネームは発生しない。これを最終防衛線とする [FO-24][R-13] |
| ID2-05 | 既定のサンプル集は `Resources/Templates/idempotency-samples.json` に持つ（20 件程度） |

## 6.2 一括リネーム（連番）[12.3 節]

```swift
public struct BulkRenamePlan: Sendable {
    public struct Item: Sendable, Identifiable {
        public let id: UUID              // ManagedFile.id または一時 ID
        public var isSelected: Bool      // [BR-03]
        public var order: Int            // D&D で入れ替え [BR-04]
        public let currentName: String
        public var newName: String       // 計算結果
        public var hasConflict: Bool     // [BR-09]
    }
    public var prefix: String            // 連番前の文字列 [BR-05]
    public var suffix: String            // 連番後の文字列 [BR-05]
    public var startNumber: Int = 1      // [BR-05]
    public var digits: Int = 2           // [BR-06]
    public var step: Int = 1             // [BR-07]
    public var items: [Item]
}

public enum BulkRenameEngine {
    public static func recompute(_ plan: inout BulkRenamePlan,
                                 existingNamesInFolder: Set<String>)
}
```

```pseudo
recompute(plan, existing):
    n = plan.startNumber
    produced = Set<String>()
    for item in plan.items.sorted(by: order) where item.isSelected:
        num = String(n).leftPad("0", to: plan.digits)
        base = plan.prefix + num + plan.suffix
        item.newName = base + "." + item.extension
        # 衝突判定: 既存ファイル（リネーム対象自身を除く）または同一実行内の重複
        item.hasConflict = (existing.contains(item.newName)
                            && item.newName != item.currentName)
                           || produced.contains(item.newName)
        produced.insert(item.newName)
        n += plan.step
```

| # | 要件 |
|---|---|
| BR2-01 | 衝突行は赤字表示し、1 件でもあれば実行ボタンを無効化する [BR-09] |
| BR2-02 | 実行は一時ファイル名を経由する **2 パス方式**。循環リネーム（A→B, B→A）に対応する [BR-10] |
| BR2-03 | 一時ファイル名は `<uuid>.qootmp`。全件を一時名にしてから最終名へ移す |
| BR2-04 | 一括リネームは 1 つの Undo 単位 [BR-11] |
| BR2-05 | 実行前に「変更前 → 変更後」の一覧をプレビューする [BR-08] |

```pseudo
convert(filename, rules, settings, styles):
    (base, ext) = splitExtension(filename)         # 拡張子は変換対象外 [CR-06]
    for r in rules where isEnabled sortedBy priority:
        if ext not in r.targetExtensions: continue
        pr = parse(base, settings.with(formats: [r.inputFormat]), .convertRename)
        if pr == nil: continue                     # 次のルールへ
        out = render(r.outputFormat, pr, styles[r.volumeStyleID], r)
        return ConvertOutcome(state, out + "." + ext, r.id, pr.spans)
    return ConvertOutcome(.unmatched, filename, nil, [])   # [CR-09]
```

```pseudo
render(template, parsed, style, rule):
    # ① 各ノードを文字列化。フィールドは値を、@volume は style で展開 [CR-21]
    rendered = renderNodes(template.nodes, parsed, style)

    # ② 欠落フィールドの処理 [CR-30〜CR-33]
    #    値が空のフィールドを含む「最も内側の区切り文字ブロック」を削除する。
    #    ブロックに囲まれていないフィールドはそのフィールドのみ削除 [CR-32]
    rendered = pruneEmptyBlocks(rendered)
    if すべてのフィールドが空: return .unmatched          # [CR-33]

    # ③ 空白の整形 [CR-31]
    s = collapseSpaces(rendered).trimmed()

    # ④ 文字の正規化・置換 [CR-40〜CR-43]
    for rep in rule.characterReplacements: s = s.replacing(rep.from, rep.to)
    if rule.normalizeWidthOnOutput: s = canonicalWidth(s)

    # ⑤ 長さ制限 [CR-44]
    if utf8Count(s + "." + ext) > AppLimits.maxFilenameUTF8Bytes:
        s = truncateFields(s, order: [.title, .series])   # 末尾から切り詰め
        warning = .filenameTruncated
    return s
```

#### `pruneEmptyBlocks` の詳細 [CR-30]

```pseudo
pruneEmptyBlocks(tree):
    # tree はレンダリング済みノード列（group はネストを保持）
    func prune(node):
        if node is .group(pair, children):
            children = children.map(prune).compactMap { $0 }
            # 直下（または子孫）のフィールドがすべて空ならブロックごと削除
            if children に含まれるフィールド由来のテキストがすべて空:
                return nil
            return .group(pair, children)
        if node is .fieldOutput(ref, text) and text.isEmpty:
            return nil                                    # [CR-32]
        return node
```

例: `[@author] @series @volume` で著者が空 → `[]` ごと削除され `@series @volume` になる [CR-30]。

## 6.4 テンポラリフォルダの処理順序 [CR-60]

```
① 自動リネーム（12.2 節、自動 / 手動）
② フォーマット変換リネーム（12.5 節、手動のみ）
③ 一括リネーム（12.3 節、手動）
④ ライブラリへの移動（12.4 節、手動）
```

この順序は UI 上のツールバー並び順・メニュー並び順にも反映する [設計判断: 手順の学習コストを下げるため]。
